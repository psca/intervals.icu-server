- [ ] **Step 1: Add a small help-link helper and `/help` page renderer in `src/index.ts`**

```ts
function helpLink(href = "/help", label = "Help"): string {
  return `<a class="page-link" href="${href}">${label}</a>`;
}

function renderHelpPage(origin: string): Response {
  const mcpUrl = `${origin}/mcp`;
  return new Response(
    `<!DOCTYPE html><html><head>
      <meta charset="utf-8">
      <title>intervals.icu MCP Help</title>
      <style>${PAGE_CSS}</style>
    </head><body>
      <div class="card docs-card">
        <div class="page-actions">${helpLink("/settings", "Settings")}</div>
        <h1>Help</h1>
        <p class="subtitle">Connect the server and find your intervals.icu credentials.</p>
        <nav class="section-nav">
          <a href="#chatgpt">ChatGPT</a>
          <a href="#claude">Claude</a>
          <a href="#intervals-icu">intervals.icu</a>
        </nav>

        <section id="chatgpt" class="doc-section">
          <h2>ChatGPT</h2>
          <p>Use this MCP URL:</p>
          <code>${escapeHtml(mcpUrl)}</code>
          <ol>
            <li>Add a new MCP connector in ChatGPT.</li>
            <li>Paste the MCP URL.</li>
            <li>Finish browser sign-in and then enter your intervals.icu credentials.</li>
          </ol>
        </section>

        <section id="claude" class="doc-section">
          <h2>Claude</h2>
          <p>Use this MCP URL:</p>
          <code>${escapeHtml(mcpUrl)}</code>
          <ol>
            <li>Add the remote MCP server in Claude.</li>
            <li>Use the `/mcp` URL above.</li>
            <li>Complete browser sign-in and then save your intervals.icu credentials.</li>
          </ol>
        </section>

        <section id="intervals-icu" class="doc-section">
          <h2>intervals.icu</h2>
          <p>You will need your athlete ID and API key.</p>
          <ol>
            <li>Open intervals.icu and go to your account settings/API area.</li>
            <li>Copy your athlete ID, which looks like <code>i12345</code>.</li>
            <li>Create or copy your API key.</li>
            <li>Paste both values into this worker when prompted.</li>
          </ol>
        </section>
      </div>
    </body></html>`,
    {
      headers: {
        "Content-Type": "text/html; charset=utf-8",
        "Content-Security-Policy": "default-src 'none'; style-src 'unsafe-inline'",
      },
    }
  );
}
```
